**Goal:** Build a testable TypeScript service in which a Corti agent investigates Karen's conversation, drafts one district-nursing blood-pressure task, waits for exact clinician approval, publishes it through MCP, routes it deterministically, and verifies or escalates it.

The UI and audio pipeline remain separate consumers of versioned HTTP/event contracts; this plan creates no UI.

import { DomainError } from "./errors.js";
import type { TaskState } from "./types.js";

const allowed: Record<TaskState, ReadonlySet<TaskState>> = {
  draft: new Set(["offered_to_team", "escalated", "dismissed"]),
  offered_to_team: new Set(["accepted", "assigned_to_member", "escalated"]),
  assigned_to_member: new Set(["accepted", "assigned_to_member", "escalated"]),
  accepted: new Set(["completed", "escalated"]),
  completed: new Set(["verified", "escalated"]),
  verified: new Set(),
  escalated: new Set(["offered_to_team", "assigned_to_member", "accepted"]),
  dismissed: new Set()
};

export function canTransition(from: TaskState, to: TaskState): boolean {
  return allowed[from].has(to);
}

export function requireTransition(from: TaskState, to: TaskState): void {
  if (!canTransition(from, to)) {
    throw new DomainError("INVALID_TRANSITION", `${from} cannot transition to ${to}`, false, 409);
  }
}

export interface Clock {

export class SystemClock implements Clock {
  now(): Date {
    return new Date();
  }
}

export class DemoClock implements Clock {
  constructor(private current: Date, private readonly enabled: boolean) {}

  now(): Date {
    return this.enabled ? new Date(this.current) : new Date();
  }

  advance(milliseconds: number): Date {
    if (!this.enabled) {
      throw new DomainError("DEMO_CLOCK_DISABLED", "Demo clock is disabled", false, 403);
    }
    if (!Number.isFinite(milliseconds) || milliseconds <= 0) {
      throw new DomainError("INVALID_CLOCK_ADVANCE", "milliseconds must be positive");
    }
    this.current = new Date(this.current.getTime() + milliseconds);
    return this.now();
  }
}

// src/domain/priority.ts
import type { ClinicalUrgency, PriorityBreakdown, Task } from "./types.js";

const BASE: Record<ClinicalUrgency, number> = { high: 100, medium: 60, routine: 20 };

function clamp(value: number, minimum: number, maximum: number): number {
  return Math.min(maximum, Math.max(minimum, value));
}

export function calculatePriority(task: Task, now: Date): PriorityBreakdown {
  const activeTargetAt = ["accepted", "completed"].includes(task.state) ? task.dueBy : task.acceptBy;
  const start = Date.parse(task.createdAt);
  const target = Date.parse(activeTargetAt);
  const current = now.getTime();
  const progress = target === start ? 1 : clamp((current - start) / (target - start), 0, 1);
  const deadlinePressure = Math.round(progress * 30);
  const overdueHours = Math.max(0, Math.floor((current - target) / 3_600_000));
  const overdue = current >= target ? Math.min(80, 40 + overdueHours * 5) : 0;
  const failedOffers = Math.min(20, task.failedOffers * 10);
  const base = BASE[task.clinicalUrgency];
  return {
    base,
    deadlinePressure,
    overdue,
    failedOffers,
    total: base + deadlinePressure + overdue + failedOffers,
    activeTargetAt
  };
}

// src/domain/routing.ts
import type { Member, Task } from "./types.js";

function hasCapabilities(member: Member, required: string[]): boolean {
  return required.every((capability) => member.capabilities.includes(capability));
}

export function chooseMember(task: Task, members: Member[]): Member | null {
  return (
    members
      .filter((member) => member.teamId === task.targetTeamId)
      .filter((member) => member.onShift && member.available)
      .filter((member) => member.openTaskCount < member.capacity)
      .filter((member) => hasCapabilities(member, task.requiredCapabilities))
      .toSorted(
        (left, right) =>
          left.openTaskCount - right.openTaskCount ||
          left.tieBreakKey.localeCompare(right.tieBreakKey) ||
          left.memberId.localeCompare(right.memberId)
      )[0] ?? null
  );
}

// src/infra/database.ts
import { mkdirSync } from "node:fs";
import path from "node:path";
import { DatabaseSync } from "node:sqlite";

export function openDatabase(databasePath: string): DatabaseSync {
  if (databasePath !== ":memory:") mkdirSync(path.dirname(databasePath), { recursive: true });
  const database = new DatabaseSync(databasePath);
  database.exec("PRAGMA foreign_keys = ON; PRAGMA journal_mode = WAL; PRAGMA busy_timeout = 5000;");
  database.exec(`
    CREATE TABLE IF NOT EXISTS patients (
      patient_id TEXT PRIMARY KEY, display_name TEXT NOT NULL, record_json TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS patient_record_items (
      item_id TEXT PRIMARY KEY, patient_id TEXT NOT NULL REFERENCES patients(patient_id),
      item_type TEXT NOT NULL, text TEXT NOT NULL, source_ref TEXT NOT NULL, recorded_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS context_mappings (
      context_id TEXT PRIMARY KEY, interaction_id TEXT NOT NULL UNIQUE,
      patient_id TEXT NOT NULL REFERENCES patients(patient_id), created_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS teams (
      team_id TEXT PRIMARY KEY, name TEXT NOT NULL, capabilities_json TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS members (
      member_id TEXT PRIMARY KEY, team_id TEXT NOT NULL REFERENCES teams(team_id),
      capabilities_json TEXT NOT NULL, on_shift INTEGER NOT NULL, available INTEGER NOT NULL,
      open_task_count INTEGER NOT NULL, capacity INTEGER NOT NULL, tie_break_key TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS threads (
      thread_id TEXT PRIMARY KEY, patient_id TEXT NOT NULL REFERENCES patients(patient_id),
      interaction_id TEXT NOT NULL, context_id TEXT, summary TEXT NOT NULL,
      evidence_refs_json TEXT NOT NULL, state TEXT NOT NULL, version INTEGER NOT NULL,
      created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS tasks (
      task_id TEXT PRIMARY KEY, thread_id TEXT NOT NULL REFERENCES threads(thread_id),
      patient_id TEXT NOT NULL REFERENCES patients(patient_id), origin TEXT NOT NULL,
      summary TEXT NOT NULL, task_type TEXT NOT NULL, evidence_refs_json TEXT NOT NULL,
      target_team_id TEXT NOT NULL REFERENCES teams(team_id), required_capabilities_json TEXT NOT NULL,
      clinical_urgency TEXT NOT NULL, operational_priority_score INTEGER NOT NULL,
      priority_breakdown_json TEXT NOT NULL, accept_by TEXT NOT NULL, due_by TEXT NOT NULL,
      state TEXT NOT NULL, assigned_member_id TEXT, failed_offers INTEGER NOT NULL,
      version INTEGER NOT NULL, created_at TEXT NOT NULL, updated_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS approvals (
      approval_id TEXT PRIMARY KEY, task_id TEXT NOT NULL REFERENCES tasks(task_id),
      patient_id TEXT NOT NULL, clinician_id TEXT NOT NULL, draft_version INTEGER NOT NULL,
      draft_hash TEXT NOT NULL, approved_at TEXT NOT NULL, approval_channel TEXT NOT NULL,
      expires_at TEXT NOT NULL, consumed_at TEXT
    );
    CREATE TABLE IF NOT EXISTS processed_commands (
      command_scope TEXT NOT NULL, idempotency_key TEXT NOT NULL, result_json TEXT NOT NULL,
      created_at TEXT NOT NULL, PRIMARY KEY (command_scope, idempotency_key)
    );
    CREATE TABLE IF NOT EXISTS task_declines (
      task_id TEXT NOT NULL REFERENCES tasks(task_id), member_id TEXT NOT NULL,
      declined_at TEXT NOT NULL, PRIMARY KEY (task_id, member_id)
    );
    CREATE TABLE IF NOT EXISTS audit_events (
      sequence INTEGER PRIMARY KEY AUTOINCREMENT, event_id TEXT NOT NULL UNIQUE,
      event_type TEXT NOT NULL, occurred_at TEXT NOT NULL, correlation_id TEXT NOT NULL,
      patient_id TEXT NOT NULL, interaction_id TEXT NOT NULL, context_id TEXT,
      actor_json TEXT NOT NULL, payload_json TEXT NOT NULL
    );
    CREATE INDEX IF NOT EXISTS idx_tasks_team_state ON tasks(target_team_id, state);
    CREATE INDEX IF NOT EXISTS idx_tasks_deadlines ON tasks(accept_by, due_by);
    CREATE INDEX IF NOT EXISTS idx_events_sequence ON audit_events(sequence);
  `);
  return database;
}

export function inTransaction<T>(database: DatabaseSync, operation: () => T): T {
  database.exec("BEGIN IMMEDIATE");
  try {
    const result = operation();
    database.exec("COMMIT");
    return result;
  } catch (error) {
    database.exec("ROLLBACK");
    throw error;
  }
}

type SqlRow = Record<string, string | number | null>;
const text = (row: SqlRow, key: string) => String(row[key]);
const optionalText = (row: SqlRow, key: string) => row[key] == null ? null : String(row[key]);
const strings = (row: SqlRow, key: string) => JSON.parse(text(row, key)) as string[];

function mapTask(row: SqlRow): Task {
  return {
    taskId: text(row, "task_id"), threadId: text(row, "thread_id"), patientId: text(row, "patient_id"),
    origin: text(row, "origin") as Task["origin"], summary: text(row, "summary"), taskType: text(row, "task_type"),

  };
}

function mapThread(row: SqlRow): Thread {
  return { threadId: text(row, "thread_id"), patientId: text(row, "patient_id"),
    interactionId: text(row, "interaction_id"), contextId: optionalText(row, "context_id"),

export class SqliteStore {
  private transactionDepth = 0;
  constructor(private readonly database: DatabaseSync) {}
  close(): void { this.database.close(); }

  transaction<T>(operation: () => T): T {
    if (this.transactionDepth > 0) return operation();
    this.transactionDepth += 1;
    try { return inTransaction(this.database, operation); }
    finally { this.transactionDepth -= 1; }
  }

  putPatient(patientId: string, displayName: string, record: unknown): void {
    this.database.prepare(`INSERT OR REPLACE INTO patients VALUES (?, ?, ?)`)
      .run(patientId, displayName, JSON.stringify(record));
  }

  getPatient(patientId: string): { patientId: string; displayName: string; record: unknown } | null {
    const row = this.database.prepare(`SELECT * FROM patients WHERE patient_id = ?`).get(patientId) as
      | { patient_id: string; display_name: string; record_json: string }
      | undefined;
    return row ? { patientId: row.patient_id, displayName: row.display_name, record: JSON.parse(row.record_json) } : null;
  }

  putRecordItem(item: PatientRecordItem): void {

      .run(item.itemId, item.patientId, item.itemType, item.text, item.sourceRef, item.recordedAt);
  }

  listRecordItems(patientId: string): PatientRecordItem[] {
    const rows = this.database.prepare(`SELECT * FROM patient_record_items WHERE patient_id = ? ORDER BY recorded_at, item_id`)
      .all(patientId) as SqlRow[];
    return rows.map((row) => ({ itemId: text(row, "item_id"), patientId: text(row, "patient_id"),
      itemType: text(row, "item_type"), text: text(row, "text"), sourceRef: text(row, "source_ref"),
      recordedAt: text(row, "recorded_at") }));
  }

  hasRecordEvidence(patientId: string, evidenceRefs: string[]): boolean {
    const known = new Set(this.listRecordItems(patientId).map((item) => item.sourceRef));
    return evidenceRefs.every((reference) => known.has(reference));

    return row ? { approvalId: text(row, "approval_id"), taskId: text(row, "task_id"), patientId: text(row, "patient_id"),
      clinicianId: text(row, "clinician_id"), draftVersion: Number(row.draft_version), draftHash: text(row, "draft_hash"),
      approvedAt: text(row, "approved_at"), approvalChannel: text(row, "approval_channel"),
      expiresAt: text(row, "expires_at"), consumedAt: optionalText(row, "consumed_at") } : null;
  }

  consumeApproval(approvalId: string, consumedAt: string): void {
    this.database.prepare(`UPDATE approvals SET consumed_at = ? WHERE approval_id = ? AND consumed_at IS NULL`).run(consumedAt, approvalId);
  }

  getProcessedCommand(scope: string, key: string): Record<string, unknown> | null {
    const row = this.database.prepare(`SELECT result_json FROM processed_commands WHERE command_scope = ? AND idempotency_key = ?`)
      .get(scope, key) as { result_json: string } | undefined;
    return row ? JSON.parse(row.result_json) as Record<string, unknown> : null;
  }

  saveProcessedCommand(scope: string, key: string, result: unknown, createdAt: string): void {
    this.database.prepare(`INSERT INTO processed_commands VALUES (?,?,?,?)`)
      .run(scope, key, JSON.stringify(result), createdAt);
  }

  runTaskCommand(scope: string, key: string, createdAt: string, operation: () => Task): Task {
    return this.transaction(() => {
      const replay = this.getProcessedCommand(scope, key);
      if (replay) {
        const task = this.getTask(String(replay.taskId));
        if (!task) throw new DomainError("TASK_NOT_FOUND", "Idempotent task result is unavailable", false, 404);
        return task;
      }
      const task = operation();
      this.saveProcessedCommand(scope, key, { taskId: task.taskId }, createdAt);
      return task;
    });
  }
}

    targetTeamId: task.targetTeamId, requiredCapabilities: task.requiredCapabilities,

  const b = Buffer.from(right);

  approveDraft(taskId: string, expectedVersion: number, clinicianId: string,
      approvalChannel = "app_one_tap",
      idempotencyKey = `approval:${taskId}:${expectedVersion}:${clinicianId}`):
      { approvalId: string; proof: string; expiresAt: string } {
    const commandScope = `approve:${taskId}:${clinicianId}`;
    const replay = this.store.getProcessedCommand(commandScope, idempotencyKey);
    if (replay) return { approvalId: String(replay.approvalId), proof: String(replay.proof),
      expiresAt: String(replay.expiresAt) };
    const task = this.store.requireTask(taskId);
    if (task.state !== "draft" || task.version !== expectedVersion) {
      throw new DomainError("VERSION_CONFLICT", "Draft changed before approval", false, 409);
    }
    const approvalId = randomUUID();
    const hash = draftHash(task);
    const approvedAt = this.clock.now().toISOString();
    const expiresAt = new Date(Date.parse(approvedAt) + 10 * 60_000).toISOString();
    const material = `${approvalId}.${hash}.${task.patientId}.${clinicianId}.${approvedAt}.${approvalChannel}.${expiresAt}`;
    const signature = createHmac("sha256", this.approvalSecret).update(material).digest("hex");
    this.store.transaction(() => {
      this.store.saveApproval({ approvalId, taskId, patientId: task.patientId, clinicianId,
        draftVersion: expectedVersion, draftHash: hash, approvedAt, approvalChannel, expiresAt, consumedAt: null });
      this.store.appendTaskEvent(task, this.store.requireThread(task.threadId).interactionId,
        this.store.requireThread(task.threadId).contextId, { type: "clinician", id: clinicianId },
        "task.approved", { approvalId, approvedAt, approvalChannel, expiresAt });
      this.store.saveProcessedCommand(commandScope, idempotencyKey,
        { approvalId, proof: `${approvalId}.${signature}`, expiresAt }, approvedAt);
    });
    return { approvalId, proof: `${approvalId}.${signature}`, expiresAt };
  }

  publishDraft(taskId: string, proof: string, expectedVersion: number, idempotencyKey: string): Task {
    const replay = this.store.getProcessedCommand(`publish:${taskId}`, idempotencyKey);
    if (replay) return this.store.requireTask(String(replay.taskId));
    return this.store.transaction(() => {
      const task = this.store.requireTask(taskId);
      if (task.version !== expectedVersion) throw new DomainError("VERSION_CONFLICT", "Draft version changed", false, 409);
      requireTransition(task.state, "offered_to_team");
      const [approvalId, signature] = proof.split(".");
      const approval = approvalId ? this.store.getApproval(approvalId) : null;
      if (!approval || !signature || approval.taskId !== taskId || approval.consumedAt ||
          approval.draftVersion !== task.version || approval.draftHash !== draftHash(task) ||
          Date.parse(approval.expiresAt) <= this.clock.now().getTime()) {
        throw new DomainError("APPROVAL_MISMATCH", "Approval does not match the current draft", false, 409);
      }
      const expected = createHmac("sha256", this.approvalSecret)
        .update(`${approval.approvalId}.${approval.draftHash}.${approval.patientId}.${approval.clinicianId}.` +
          `${approval.approvedAt}.${approval.approvalChannel}.${approval.expiresAt}`).digest("hex");
      if (!safeEqual(signature, expected)) throw new DomainError("APPROVAL_MISMATCH", "Approval proof is invalid", false, 409);
      const updated = { ...task, state: "offered_to_team" as const, version: task.version + 1,
        updatedAt: this.clock.now().toISOString() };
      this.store.replaceTask(task, updated);
      this.store.consumeApproval(approval.approvalId, updated.updatedAt);
      this.store.saveProcessedCommand(`publish:${taskId}`, idempotencyKey, { taskId }, updated.updatedAt);
      const thread = this.store.requireThread(task.threadId);
      this.store.setThreadState(thread.threadId, thread.version, "tracking", updated.updatedAt);
      this.store.appendTaskEvent(updated, thread.interactionId, thread.contextId,
        { type: "agent", id: "corti" }, "task.published_to_team", { approvalId });
      return updated;
    });
  }

  tick(): void {
    const now = this.clock.now();
    for (const task of this.store.listNonTerminalTasks()) {
      const priorityBreakdown = calculatePriority(task, now);
      this.store.refreshPriority(task.taskId, task.version, priorityBreakdown, now.toISOString());
      const current = this.store.requireTask(task.taskId);

      if (current.state === "offered_to_team" && Date.parse(current.acceptBy) <= now.getTime()) {
        const declined = new Set(this.store.listDeclinedMemberIds(current.taskId));
        const member = chooseMember(current, this.store.listMembers(current.targetTeamId).filter((candidate) => !declined.has(candidate.memberId)));
        if (member) this.store.assignMember(current.taskId, current.version, member.memberId, now.toISOString());
        else this.store.escalate(current.taskId, current.version, "NO_ELIGIBLE_MEMBER", now.toISOString());
        continue;
      }

      if (["draft", "assigned_to_member", "accepted", "completed"].includes(current.state) &&
          Date.parse(current.dueBy) <= now.getTime()) {
        this.store.escalate(current.taskId, current.version, "DUE_WITHOUT_VERIFICATION", now.toISOString());
      }
    }
  }

  decline(taskId: string, expectedVersion: number, memberId: string): void {
    const now = this.clock.now().toISOString();
    const task = this.store.recordDecline(taskId, expectedVersion, memberId, now);
    const declined = new Set(this.store.listDeclinedMemberIds(taskId));
    const next = chooseMember(task, this.store.listMembers(task.targetTeamId).filter((candidate) => !declined.has(candidate.memberId)));
    if (next) this.store.reassignMember(taskId, task.version, next.memberId, now);
    else this.store.escalate(taskId, task.version, "ALL_ELIGIBLE_MEMBERS_DECLINED", now);
  }
}

  assignMember(taskId: string, expectedVersion: number, memberId: string, updatedAt: string): Task {
    return this.transaction(() => {
      const current = this.requireTask(taskId);
      if (current.version !== expectedVersion || current.state !== "offered_to_team") {
        throw new DomainError("VERSION_CONFLICT", "Task changed before timeout assignment", false, 409);
      }
      const member = this.requireEligibleMember(memberId, current);
      const next = { ...current, state: "assigned_to_member" as const, assignedMemberId: member.memberId,
        failedOffers: current.failedOffers + 1, version: current.version + 1, updatedAt };
      this.replaceTask(current, next);
      const thread = this.requireThread(next.threadId);
      this.appendTaskEvent(next, thread.interactionId, thread.contextId, { type: "router", id: "timeout-router" },
        "task.team_acceptance_timed_out", { acceptBy: current.acceptBy });
      this.appendTaskEvent(next, thread.interactionId, thread.contextId, { type: "router", id: "timeout-router" },
        "task.member_assigned", { memberId });
      return next;
    });
  }

  escalate(taskId: string, expectedVersion: number, reason: string, updatedAt: string): Task {
    return this.transaction(() => {
      const current = this.requireTask(taskId);
      if (current.version !== expectedVersion) throw new DomainError("VERSION_CONFLICT", "Task changed before escalation", false, 409);
      if (["verified", "dismissed", "escalated"].includes(current.state)) return current;
      const next = { ...current, state: "escalated" as const, version: current.version + 1, updatedAt };
      this.replaceTask(current, next);
      const thread = this.requireThread(next.threadId);
      this.appendTaskEvent(next, thread.interactionId, thread.contextId, { type: "router", id: "deadline-policy" },
        "task.escalated", { reason });
      this.setThreadState(thread.threadId, thread.version, "escalated", updatedAt);
      return next;
    });
  }

  listDeclinedMemberIds(taskId: string): string[] {
    const rows = this.database.prepare(`SELECT member_id FROM task_declines WHERE task_id = ? ORDER BY declined_at`)
      .all(taskId) as Array<{ member_id: string }>;
    return rows.map((row) => row.member_id);
  }

  recordDecline(taskId: string, expectedVersion: number, memberId: string, declinedAt: string): Task {
    return this.transaction(() => {
      const current = this.requireTask(taskId);
      if (current.version !== expectedVersion || current.state !== "assigned_to_member" || current.assignedMemberId !== memberId) {
        throw new DomainError("VERSION_CONFLICT", "Assignment changed before decline", false, 409);
      }
      this.database.prepare(`INSERT OR IGNORE INTO task_declines VALUES (?,?,?)`).run(taskId, memberId, declinedAt);
      const next = { ...current, assignedMemberId: null, failedOffers: current.failedOffers + 1,
        version: current.version + 1, updatedAt: declinedAt };
      this.replaceTask(current, next);
      const thread = this.requireThread(next.threadId);
      this.appendTaskEvent(next, thread.interactionId, thread.contextId, { type: "team_member", id: memberId },
        "task.member_declined", { memberId });
      return next;
    });
  }

  reassignMember(taskId: string, expectedVersion: number, memberId: string, assignedAt: string): Task {
    return this.updateTask(taskId, expectedVersion, (task) => ({ ...task, assignedMemberId: memberId,
      version: task.version + 1, updatedAt: assignedAt }), { type: "router", id: "decline-router" },
      "task.member_assigned", { memberId });
  }

import { DomainError } from "../domain/errors.js";
import type { SqliteStore } from "../infra/store.js";

export class RecordService {
  constructor(private readonly store: SqliteStore) {}

  private requirePatient(contextId: string, requestedPatientId: string): void {
    const scopedPatientId = this.store.patientForContext(contextId);
    if (!scopedPatientId || scopedPatientId !== requestedPatientId) {
      throw new DomainError("PATIENT_SCOPE_DENIED", "Patient scope is unavailable", false, 403);
    }
  }

  requireInteraction(contextId: string, interactionId: string): void {
    if (this.store.contextForInteraction(interactionId) !== contextId) {
      throw new DomainError("CONTEXT_INTERACTION_MISMATCH", "Interaction scope is unavailable", false, 403);
    }
  }

  getPatientContext(contextId: string, patientId: string) {
    this.requirePatient(contextId, patientId);
    const patient = this.store.getPatient(patientId);
    if (!patient) throw new DomainError("RECORD_LOOKUP_FAILED", "Patient record could not be loaded", true, 503);
    return { ...patient, recordItems: this.store.listRecordItems(patientId) };
  }

  listOpenThreads(contextId: string, patientId: string) {
    this.requirePatient(contextId, patientId);
    return this.store.listOpenThreads(patientId);
  }

  listEligibleTeams(contextId: string, patientId: string, requiredCapabilities: string[]) {
    this.requirePatient(contextId, patientId);
    return this.store.listTeams()
      .filter((team) => requiredCapabilities.every((capability) => team.capabilities.includes(capability)))
      .map((team) => {
        const members = this.store.listMembers(team.teamId);
        return { ...team, availability: {
          onShift: members.filter((member) => member.onShift).length,
          availableWithCapacity: members.filter((member) => member.onShift && member.available &&
            member.openTaskCount < member.capacity).length
        } };
      });
  }
}

import { DomainError } from "../domain/errors.js";

function success(value: unknown) {
  return { content: [{ type: "text" as const, text: JSON.stringify(value) }], structuredContent: value as Record<string, unknown> };
}

function failure(error: unknown) {
  const domain = error instanceof DomainError ? error : new DomainError("INTERNAL_ERROR", "Tool failed", true, 500);
  return { content: [{ type: "text" as const, text: JSON.stringify({ code: domain.code, message: domain.message, retryable: domain.retryable }) }], isError: true };

export function createFollowThroughMcp(records: RecordService, ledger: LedgerService, store: SqliteStore): McpServer {

  server.registerTool("get_patient_context", {
    description: "Retrieve scoped synthetic record facts for one patient. Never infer an empty record from an error.",
    inputSchema: { patientId: z.string().min(1) }
  }, async ({ patientId }, extra) => {
    try {
      const contextId = contextIdFromMeta(extra);
      const result = records.getPatientContext(contextId, patientId);
      store.appendContextEvent(contextId, "record.context_retrieved", { evidenceAvailable: true });
      return success(result);
    } catch (error) { return failure(error); }
  });

  server.registerTool("list_open_threads", {
    description: "List active follow-through threads for duplicate checking.",
    inputSchema: { patientId: z.string().min(1) }
  }, async ({ patientId }, extra) => {
    try {
      const contextId = contextIdFromMeta(extra);
      const result = records.listOpenThreads(contextId, patientId);
      store.appendContextEvent(contextId, "record.open_threads_checked", { count: result.length });
      return success({ threads: result });
    } catch (error) { return failure(error); }
  });

  server.registerTool("list_eligible_teams", {
    description: "List eligible teams only; never choose an individual person.",
    inputSchema: { patientId: z.string().min(1), requiredCapabilities: z.array(z.string().min(1)).min(1) }
  }, async ({ patientId, requiredCapabilities }, extra) => {
    try { return success({ teams: records.listEligibleTeams(contextIdFromMeta(extra), patientId, requiredCapabilities) }); }
    catch (error) { return failure(error); }
  });

  server.registerTool("create_task_draft", {
    description: "Create a non-actionable proposed team task. A likely duplicate is an error requiring clinician review.",
    inputSchema: {
      patientId: z.string().min(1), interactionId: z.string().min(1), threadId: z.string().uuid().optional(),
      summary: z.string().min(5).max(240), taskType: z.string().min(1), evidenceRefs: z.array(z.string()).min(1),
      targetTeamId: z.string().min(1), requiredCapabilities: z.array(z.string()).min(1),
      clinicalUrgency: z.enum(["high", "medium", "routine"]), dueInMs: z.number().int().positive(),
      idempotencyKey: z.string().min(8)
    }
  }, async (input, extra) => {
    try {
      const contextId = contextIdFromMeta(extra);
      records.getPatientContext(contextId, input.patientId);
      records.requireInteraction(contextId, input.interactionId);
      return success(ledger.createDraft({ ...input, contextId, origin: "agent_suggested",
        actor: { type: "agent", id: "corti" } }));
    } catch (error) { return failure(error); }
  });

  server.registerTool("publish_team_task", {
    description: "Publish one exact clinician-approved draft to its team. Requires approval proof, version, and idempotency key.",
    inputSchema: { patientId: z.string().min(1), taskId: z.string().uuid(), approvalProof: z.string().min(10), expectedVersion: z.number().int().positive(), idempotencyKey: z.string().min(8) }
  }, async ({ patientId, taskId, approvalProof, expectedVersion, idempotencyKey }, extra) => {
    try {
      records.getPatientContext(contextIdFromMeta(extra), patientId);
      if (ledger.getTask(taskId).patientId !== patientId) {
        throw new DomainError("PATIENT_SCOPE_DENIED", "Patient scope is unavailable", false, 403);
      }
      return success(ledger.publishDraft(taskId, approvalProof, expectedVersion, idempotencyKey));
    } catch (error) { return failure(error); }
  });

        store.appendTaskEvent(task, thread.interactionId, thread.contextId,

      }

    } catch (error) { return failure(error); }
  });

  return server;
}

```ts
// src/mcp/transport.ts
import { randomUUID } from "node:crypto";
import { StreamableHTTPServerTransport } from "@modelcontextprotocol/sdk/server/streamableHttp.js";
import { isInitializeRequest } from "@modelcontextprotocol/sdk/types.js";
import type { Request, Response, Router } from "express";
import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { hasBearer } from "./auth.js";

export function mountMcp(router: Router, createServer: () => McpServer, bearerToken: string): void {
  const sessions = new Map<string, StreamableHTTPServerTransport>();

  const failTransport = (response: Response): void => {
    if (!response.headersSent) {
      response.status(500).json({ jsonrpc: "2.0", error: { code: -32603, message: "MCP transport failed" }, id: null });
    } else response.end();
  };

  const handlePost = async (request: Request, response: Response) => {
    if (request.body?.method === "tools/call" && !hasBearer(request, bearerToken)) {
      response.status(401).json({ jsonrpc: "2.0", error: { code: -32001, message: "Unauthorized" }, id: request.body?.id ?? null });
      return;
    }
    const sessionId = request.header("mcp-session-id");
    let transport = sessionId ? sessions.get(sessionId) : undefined;
    if (!transport && !sessionId && isInitializeRequest(request.body)) {
      transport = new StreamableHTTPServerTransport({
        sessionIdGenerator: () => randomUUID(),
        onsessioninitialized: (id) => sessions.set(id, transport as StreamableHTTPServerTransport)
      });
      transport.onclose = () => { if (transport?.sessionId) sessions.delete(transport.sessionId); };
      await createServer().connect(transport);
    }
    if (!transport) {
      response.status(400).json({ jsonrpc: "2.0", error: { code: -32000, message: "Invalid MCP session" }, id: request.body?.id ?? null });
      return;
    }
    await transport.handleRequest(request, response, request.body);
  };
  router.post("/mcp", (request, response) => { void handlePost(request, response).catch(() => failTransport(response)); });

  const existing = async (request: Request, response: Response) => {
    const transport = sessions.get(request.header("mcp-session-id") ?? "");
    if (!transport) { response.status(400).send("Invalid MCP session"); return; }
    await transport.handleRequest(request, response);
  };
  router.get("/mcp", (request, response) => { void existing(request, response).catch(() => failTransport(response)); });
  router.delete("/mcp", (request, response) => { void existing(request, response).catch(() => failTransport(response)); });
}

import type { NextFunction, Request, Response } from "express";
import { DomainError } from "../domain/errors.js";
import { hasBearer } from "../mcp/auth.js";

export function requireAppAuth(expected: string) {
  return (request: Request, response: Response, next: NextFunction): void => {
    if (!hasBearer(request, expected)) { response.status(401).json({ error: { code: "UNAUTHORIZED" } }); return; }
    next();
  };
}

export function requireActor(request: Request): string {
  const actorId = request.header("x-actor-id");
  if (!actorId) throw new DomainError("ACTOR_REQUIRED", "x-actor-id is required");
  return actorId;
}

# Follow-Through integration contract v1

This is the frozen MVP contract between the pipeline, UI, and Agentic/MCP backend. All examples use synthetic data.

## Authentication and attribution

- `GET /healthz` is public.
- Every `/api/*` request requires `Authorization: Bearer $APP_BEARER_TOKEN`.
- Every `/api/*` mutation also requires `x-actor-id`.
- Every mutation body carries an operation-specific `idempotencyKey`; retry with the same key and payload.
- MCP `tools/call` requests require `Authorization: Bearer $MCP_BEARER_TOKEN`.
- Browser access is allowed only from the configured `UI_ORIGIN`; production should place the app credential behind the UI's server/BFF.
- Corti, MCP, HMAC, and tunnel credentials are server-side only and never appear in events or ordinary Corti DataParts.

Errors use:

```json
{"error":{"code":"VERSION_CONFLICT","message":"Task changed concurrently","retryable":false}}
```

Consumers must ignore additive fields they do not recognize and reject any event whose major `schemaVersion` is not `"1"`.

## Pipeline command

`POST /api/signals`

```json
{"patientId":"synthetic-karen","interactionId":"interaction-karen-1","signalText":"Dizziness since medication change; no BP follow-up owner","evidenceRefs":["encounter:sentence-42"],"idempotencyKey":"signal-karen-001"}
```

With the runner disabled, `202` is:

```json
{"signalEventId":"event-uuid","status":"retained","recovery":"MANUAL_TASK_AVAILABLE"}
```

With the Corti runner enabled, `202` is:

```json
{"signalEventId":"event-uuid","contextId":"corti-context-id","cortiTaskId":"corti-task-id","agentState":"completed","credits":0.0042}
```

If Corti fails or exceeds the bounded wait, `502` retains the signal and returns `AGENT_INVESTIGATION_FAILED` with `recovery: "MANUAL_TASK_AVAILABLE"`; it never claims that no concern exists.

## Clinician commands

`POST /api/tasks/manual` creates a draft through the same ledger:

```json
{"patientId":"synthetic-karen","interactionId":"interaction-karen-1","contextId":"ctx-karen","summary":"Check blood pressure within 48 hours","taskType":"blood-pressure-check","evidenceRefs":["dictation:manual-1"],"targetTeamId":"district-nursing","requiredCapabilities":["blood-pressure"],"clinicalUrgency":"medium","dueInMs":172800000,"idempotencyKey":"manual-karen-001"}
```

It returns `201` with the `Task` object defined below in `draft` state.
When Corti never established a context, omit `contextId`; the authenticated clinician-created draft remains available and stores a null Corti context without weakening patient validation in MCP calls.

`POST /api/tasks/11111111-1111-4111-8111-111111111111/correct` accepts one or more corrections:

```json
{"expectedVersion":1,"summary":"Check seated and standing BP within 48 hours","targetTeamId":"district-nursing","requiredCapabilities":["blood-pressure"],"clinicalUrgency":"medium","dueInMs":172800000,"idempotencyKey":"correct-karen-001"}
```

It returns the corrected `Task`; its version increments and every earlier approval becomes invalid.

`POST /api/tasks/11111111-1111-4111-8111-111111111111/approve`:

```json
{"expectedVersion":2,"idempotencyKey":"approve-karen-001"}
```

When Corti publication is enabled, `200` is:

```json
{"task":{"taskId":"11111111-1111-4111-8111-111111111111","state":"offered_to_team","version":3},"contextId":"ctx-karen","cortiTaskId":"corti-publish-task","agentState":"completed","credits":0.0021}
```

When the runner is unavailable, `200` preserves the human decision without pretending to publish:

```json
{"taskId":"11111111-1111-4111-8111-111111111111","approvalProof":"approval-id.hmac-signature","expiresAt":"2026-08-20T10:10:00.000Z","status":"approved_not_published"}
```

`POST /api/tasks/11111111-1111-4111-8111-111111111111/dismiss`:

```json
{"expectedVersion":1,"reason":"Already covered by an existing order","idempotencyKey":"dismiss-karen-001"}
```

It returns the `Task` in `dismissed` state.

`POST /api/tasks/11111111-1111-4111-8111-111111111111/reopen` is an explicit clinician recovery for an escalated task:

```json
{"expectedVersion":6,"dueInMs":172800000,"idempotencyKey":"reopen-karen-001"}
```

It returns the `Task` in `offered_to_team` state with a fresh acceptance window and explicit clinician-supplied due interval; it does not alter clinical urgency.

{"expectedVersion":3,"idempotencyKey":"accept-karen-nurse-a-001"}

{"milliseconds":1860000,"idempotencyKey":"clock-karen-31m-001"}
```

## Queries

- `GET /healthz` returns `{"ok":true}`.
- `GET /api/patients/synthetic-karen/threads` returns `{"threads":[Thread]}`.
- `GET /api/patients/synthetic-karen/tasks` returns `{"tasks":[Task]}` including drafts for clinician review.
- `GET /api/teams/district-nursing/tasks` returns `{"tasks":[Task]}`, sorted by operational score then deadline.
- `GET /api/tasks/11111111-1111-4111-8111-111111111111` returns one authoritative `Task`.
- `GET /api/events?after=42` returns `{"events":[DomainEvent]}` with sequence values greater than 42.
- `GET /api/events/stream` is SSE. Reconnect with `Last-Event-ID: 42`; clients must also refresh authoritative task state after reconnecting.

## Shared objects

`Task` contains `taskId`, `threadId`, `patientId`, `origin`, `summary`, `taskType`, `evidenceRefs`, `targetTeamId`, `requiredCapabilities`, clinician-owned `clinicalUrgency` (correctable while still a draft), dynamic `operationalPriorityScore`, `priorityBreakdown`, `acceptBy`, `dueBy`, `state`, nullable `assignedMemberId`, `failedOffers`, `version`, `createdAt`, and `updatedAt`.

Task states are `draft`, `offered_to_team`, `assigned_to_member`, `accepted`, `completed`, `verified`, `escalated`, and `dismissed`. Thread states are `awaiting_review`, `tracking`, `verified`, `escalated`, and `dismissed`.

Every event is:

```json
{"schemaVersion":"1","sequence":43,"eventId":"event-uuid","eventType":"task.published_to_team","occurredAt":"2026-08-20T10:02:00.000Z","correlationId":"thread-uuid","patientId":"synthetic-karen","interactionId":"interaction-karen-1","contextId":"ctx-karen","actor":{"type":"agent","id":"corti"},"payload":{"taskId":"11111111-1111-4111-8111-111111111111","threadId":"thread-uuid","state":"offered_to_team","version":3}}
```

// src/agent/runner.ts
import type { SqliteStore } from "../infra/store.js";

export interface AgentSendInput { text: string; contextId?: string; data?: unknown; }
export interface AgentResult { contextId: string; taskId: string | null; state: string; credits?: number; }
export interface AgentGateway {
  send(input: AgentSendInput): Promise<AgentResult>;
  waitForCompletion(result: AgentResult): Promise<AgentResult>;
}

export class AgentRunner {
  constructor(private readonly gateway: AgentGateway, private readonly store: SqliteStore, private readonly mcpToken: string) {}

  private async ensureContext(patientId: string, interactionId: string): Promise<string> {
    const existing = this.store.contextForInteraction(interactionId);
    if (existing) return existing;
    const warmup = await this.gateway.send({ text: "Initialize this interaction context. Do not call tools." });
    this.store.putContextMapping(warmup.contextId, interactionId, patientId, new Date().toISOString());
    return warmup.contextId;
  }

  async investigate(input: { patientId: string; interactionId: string; signalText: string; evidenceRefs: string[] }) {
    const contextId = await this.ensureContext(input.patientId, input.interactionId);
    this.store.appendContextEvent(contextId, "agent.investigation_started", { evidenceRefs: input.evidenceRefs });
    const result = await this.gateway.send({
      contextId,
      text: `Investigate this patient-scoped follow-through signal: ${input.signalText}`,
      data: { patientId: input.patientId, interactionId: input.interactionId, evidenceRefs: input.evidenceRefs,
        mcpToken: this.mcpToken }
    });
    return this.gateway.waitForCompletion(result);
  }

  async publishApproved(input: { patientId: string; interactionId: string; taskId: string; expectedVersion: number; approvalProof: string; idempotencyKey: string }) {
    const contextId = await this.ensureContext(input.patientId, input.interactionId);
    const result = await this.gateway.send({ contextId,
      text: "Publish the exact approved task with publish_team_task, then verify it with get_task.",
      data: { ...input, mcpToken: this.mcpToken } });
    return this.gateway.waitForCompletion(result);
  }
}

```ts
// src/agent/corti-gateway.ts
import { randomUUID } from "node:crypto";
import { CortiClient, type Corti } from "@corti/sdk";
import type { AppConfig } from "../config.js";
import type { AgentGateway, AgentResult, AgentSendInput } from "./runner.js";

export class CortiSdkGateway implements AgentGateway {
  private readonly client: CortiClient;
  private readonly mcpName: string;
  constructor(private readonly agentId: string, config: AppConfig) {
    this.mcpName = config.mcpName;
    this.client = new CortiClient({ environment: config.corti.environment, tenantName: config.corti.tenantName,
      auth: { clientId: config.corti.clientId, clientSecret: config.corti.clientSecret },
      analytics: { examples_repo: "team-13-follow-through" } });
  }

  async send(input: AgentSendInput): Promise<AgentResult> {
    const parts: Corti.AgentsPart[] = [{ kind: "text", text: input.text }];
    if (input.data) {
      const data = input.data as Record<string, unknown>;
      const mcpToken = data.mcpToken;
      const safeData = { ...data };
      delete safeData.mcpToken;
      parts.push({ kind: "data", data: safeData });
      if (typeof mcpToken === "string") parts.push({ kind: "data", data: { type: "token", mcp_name: this.mcpName, token: mcpToken } });
    }
    const message: Corti.AgentsMessage = { role: "user", kind: "message", messageId: randomUUID(), parts };
    if (input.contextId) message.contextId = input.contextId;
    const response = await this.client.agents.messageSend(this.agentId,
      { message, configuration: { blocking: false } }, { timeoutInSeconds: 60, maxRetries: 0 });
    const value = response.task ?? response.message;
    if (!value?.contextId) throw new Error("Corti response omitted contextId");
    const credits = response.task?.metadata?.$usage?.credits;
    return { contextId: value.contextId, taskId: response.task?.id ?? null,
      state: response.task?.status.state ?? "completed", ...(credits === undefined ? {} : { credits }) };
  }

  async waitForCompletion(result: AgentResult): Promise<AgentResult> {
    if (!result.taskId || result.state === "completed") return result;
    if (["failed", "canceled", "rejected"].includes(result.state)) {
      throw new Error(`Corti agent ended in ${result.state}`);
    }
    const deadline = Date.now() + 180_000;
    while (Date.now() < deadline) {
      await new Promise((resolve) => setTimeout(resolve, 1_000));
      const task = await this.client.agents.getTask(this.agentId, result.taskId);
      if (["completed", "failed", "canceled", "rejected"].includes(task.status.state)) {
        const credits = task.metadata?.$usage?.credits;
        const terminal = { contextId: task.contextId, taskId: task.id, state: task.status.state,
          ...(credits === undefined ? {} : { credits }) };
        if (terminal.state !== "completed") throw new Error(`Corti agent ended in ${terminal.state}`);
        return terminal;
      }
    }
    throw new Error("Corti agent stalled for 180 seconds");
  }
}
